cmake_minimum_required(VERSION 3.10)
project(n3mapping_core LANGUAGES CXX)
set(N3MAPPING_ROOT ${CMAKE_CURRENT_SOURCE_DIR})
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
add_compile_definitions(N3MAPPING_SOURCE_DIR="${N3MAPPING_ROOT}")
include(cmake/n3mapping_dependencies.cmake)

include(cmake/n3mapping_core.cmake)
add_library(n3mapping_core SHARED ${N3MAPPING_CORE_SOURCES})
n3mapping_configure_core_target(n3mapping_core)
include(cmake/n3mapping_nav_resource_reader.cmake)
n3mapping_add_nav_resource_reader(n3map_nav_resource_reader)
include(CTest)
if(BUILD_TESTING)
  find_package(GTest REQUIRED)
  foreach(check IN ITEMS test_realtime_output test_global_map_cache)
    add_executable(${check} test/${check}.cpp)
    target_include_directories(${check} PRIVATE ${JSONCPP_INCLUDE_DIRS})
    target_link_libraries(${check} PRIVATE n3mapping_core GTest::GTest GTest::Main ${JSONCPP_LIBRARIES})
    add_test(NAME ${check} COMMAND ${check})
  endforeach()
  add_test(NAME core_no_ros_includes COMMAND ${CMAKE_COMMAND}
    -DROOT_DIR=${N3MAPPING_ROOT} -P ${N3MAPPING_ROOT}/test/check_core_no_ros_includes.cmake)
endif()
